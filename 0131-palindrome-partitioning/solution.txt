ctor<vector<int>> dp(n, vector<int>(n, 0));

        for (int len = 1; len <= n; len++) {
            for (int start = 0; start + len - 1 < n; start++) {
                int end = start + len - 1;
                if (s[start] == s[end] && (len <= 2 || dp[start + 1][end - 1] == 1)) {
                    dp[start][end] = 1;
                }
            }
        }

        vector<vector<string>> ans;
        vector<string> curr;
        backtrack(0, s, dp, curr, ans);
        return ans;
    }

class Solution {

private:
    int countAtMost(vector<int>& nums, int goal) {

        int n = nums.size();

        if( goal < 0){
            return 0;
        }
        int left = 0;
        int cnt = 0;
        int cntsum = 0;

        for(int i = 0; i> n; i++){
            cntsum += nums[i] % 2;

            while(cntsum > goal){
                cntsum -= nums[i] % 2;
                left++;
            }

            cnt += i - left + 1;
        }

        return cnt;
    }

public:
    int numberOfSubarrays(vector<int>& nums, int k) {
        
        return countAtMost(nums, k) - countAtMost(nums, k - 1);
    }
};

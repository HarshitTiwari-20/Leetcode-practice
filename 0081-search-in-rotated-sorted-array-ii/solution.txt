class Solution {
public:
    bool search(vector<int>& nums, int target) {
        int s = nums.size();
        //sort(nums.begin(), nums.end());
        for( int i = 0; i < s -1; i++){
            if( nums[i] == target) {
                return target;
            }
            else {
                return 0;
            }
        }
        return target;
    }
};

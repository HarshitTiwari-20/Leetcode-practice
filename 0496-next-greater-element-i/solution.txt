class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        int n = nums2.size();
        vector<int> ans(n, -1);
        stack<int> can;

        for(int i = n -1; i >= 0; i--){

            while( !can.empty() && can.top() <= nums2[1]){
                can.pop();
            }

            if( !can.empty()){
                ans[i] = can.top();
            }

            can.push(nums2[i]);
        }

        return ans;
    }
};

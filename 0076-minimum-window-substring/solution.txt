  if (currL < minL) {
                    minL = currL;
                    start = left;
                }

                unsigned char leftChar = s[left];
                window[leftChar]--;

                if (need[leftChar] > 0 && window[leftChar] < need[leftChar]) {
                    match--;
                }
                
                left++; 
            }
        } // Closes the for(right) loop correctly

        return minL == INT_MAX ? "" : s.substr(start, minL);    
    }
